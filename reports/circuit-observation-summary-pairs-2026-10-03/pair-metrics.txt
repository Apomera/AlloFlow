    async function pairState(name,width,scale){
      const groups=await summaries.locator('.circuit-notebook-observation > summary').evaluateAll(elements=>{const rect=el=>{const r=el.getBoundingClientRect();return {left:r.left,right:r.right,top:r.top,bottom:r.bottom,width:r.width,height:r.height};};return elements.map((summary,index)=>({index,actualViewportWidth:innerWidth,summaryDirection:getComputedStyle(summary).direction,pairs:[...summary.querySelectorAll('.circuit-observation-summary-probe-pair')].map(pair=>{const label=pair.querySelector('.circuit-observation-summary-probe-label'),node=pair.querySelector('.circuit-observation-summary-probe-node'),s=getComputedStyle(pair);return {lead:pair.dataset.probeLead,dir:pair.dir,direction:s.direction,bidi:s.unicodeBidi,display:s.display,text:pair.textContent,label:label.textContent,node:node.textContent,nodeDir:node.dir,pairRect:rect(pair),labelRect:rect(label),nodeRect:rect(node)};})}));});
      const pointRecords=results.summaryStates.at(-1).points;assert.equal(groups.length,pointRecords.length);
      for(const group of groups){assert.equal(group.actualViewportWidth,width);assert.deepEqual(group.pairs.map(p=>p.lead),['red','black']);for(const pair of group.pairs){assert.equal(pair.dir,'auto');assert.equal(pair.direction,'ltr');assert.ok(pair.bidi.includes('isolate'));assert.equal(pair.display,'inline-block');assert.equal(pair.nodeDir,'ltr');assert.equal(pair.node,pointRecords[group.index][pair.lead==='red'?'probeRed':'probeBlack']);assert.equal(pair.text,pair.label+' '+pair.node);assert.ok(pair.nodeRect.left>=pair.labelRect.left-1);for(const part of [pair.labelRect,pair.nodeRect]){assert.ok(part.left>=pair.pairRect.left-1);assert.ok(part.right<=pair.pairRect.right+1);assert.ok(part.top>=pair.pairRect.top-1);assert.ok(part.bottom<=pair.pairRect.bottom+1);}}
        const [red,black]=group.pairs;if(group.summaryDirection==='rtl'&&Math.abs(red.pairRect.top-black.pairRect.top)<1)assert.ok(red.pairRect.right>black.pairRect.right,'RTL red pair must precede black pair as a whole');
      }results.probePairs.push({name,width,scale,groups});
      const accessible=[];results.accessibleSummaries.push({name,width,scale,entries:accessible});const cdp=await page.context().newCDPSession(page);
      try{await cdp.send('Accessibility.enable');const {root:documentRoot}=await cdp.send('DOM.getDocument',{depth:0});for(let i=0;i<pointRecords.length;i++){const snapshot=await summaries.locator('.circuit-notebook-observation > summary').nth(i).ariaSnapshot(),point=pointRecords[i],pairs=groups[i].pairs;const {nodeId}=await cdp.send('DOM.querySelector',{nodeId:documentRoot.nodeId,selector:'#root .circuit-notebook-observations > .circuit-notebook-observation:nth-child('+(i+1)+') > summary'});assert.ok(nodeId);const {nodes}=await cdp.send('Accessibility.getPartialAXTree',{nodeId,fetchRelatives:false}),node=nodes[0],nativeName=node.name?.value||'',role=node.role?.value;accessible.push({index:i,snapshot,role,nativeName,ignored:node.ignored});assert.equal(node.ignored,false);assert.equal(role,'DisclosureTriangle');assert.ok(nativeName.includes(String(i+1).padStart(2,'0')+' '+point.project),'Accessible title needs a word separator');assert.ok(nativeName.includes(point.label+' '+point.current),'Accessible current label and value need a word separator');for(const pair of pairs)assert.ok(nativeName.includes(pair.label+' '+pair.node),'Accessible probe pair needs a word separator');}}finally{await cdp.detach();}
    }

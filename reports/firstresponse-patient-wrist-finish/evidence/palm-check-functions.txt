function wristTriangleSample(wrist,lane,row,t,normal){
  const u=.37,around=wrist.geometry.parameters.radialSegments,index=wrist.geometry.index.array,offset=(row*around+lane)*6+(u+t<=1?0:3),weights=u+t<=1?[1-u-t,u,t]:[1-t,t+u-1,1-u],attribute=wrist.geometry.attributes[normal?'normal':'position'],matrix=normal?new THREE.Matrix3().getNormalMatrix(wrist.matrixWorld):wrist.matrixWorld,result=new THREE.Vector3();
  for(let corner=0;corner<3;corner++){const value=new THREE.Vector3().fromBufferAttribute(attribute,index[offset+corner]);if(normal)value.applyMatrix3(matrix).normalize();else value.applyMatrix4(matrix);result.addScaledVector(value,weights[corner]);}return normal?result.normalize():result;
}
function palmCrossings(wrist,palm,forearm){
  const around=wrist.geometry.parameters.radialSegments,rings=wrist.geometry.parameters.ringCount,position=wrist.geometry.attributes.position;
  const center=palm.getWorldPosition(new THREE.Vector3()),scale=new THREE.Vector3().setFromMatrixScale(wrist.parent.matrixWorld).x,ray=new THREE.Raycaster(),records=[];
  const side=palm.material.side;palm.material.side=THREE.DoubleSide;
  function edgePoint(lane,row,t){return wristTriangleSample(wrist,lane,row,t,false);}
  function surface(point){const direction=point.clone().sub(center).normalize();ray.set(center,direction);const hit=ray.intersectObject(palm,false)[0];check(!!hit,'actual sculpted palm intersects radial surface query '+JSON.stringify({palm:palm.name,point:point.toArray(),center:center.toArray(),direction:direction.toArray()}));return{hit,gap:(point.distanceTo(center)-hit.distance)/scale};}
  try{for(let lane=0;lane<around;lane++)for(let row=7;row<rings-1;row++){
    const a=surface(edgePoint(lane,row,0)),b=surface(edgePoint(lane,row,1));if(!(a.gap>=0&&b.gap<0))continue;
    let lo=0,hi=1;for(let step=0;step<20;step++){const t=(lo+hi)/2;if(surface(edgePoint(lane,row,t)).gap>=0)lo=t;else hi=t;}
    const t=(lo+hi)/2,point=edgePoint(lane,row,t),s=surface(point),matrix=new THREE.Matrix3().getNormalMatrix(wrist.matrixWorld),normalAttribute=wrist.geometry.attributes.normal;
    const wristNormal=wristTriangleSample(wrist,lane,row,t,true),palmNormal=shaderNormal(s.hit);
    const outward=wristNormal.clone().add(palmNormal).normalize();ray.set(point.clone().addScaledVector(outward,.12*scale),outward.clone().negate());
    const blocker=ray.intersectObjects([wrist,palm,forearm],false)[0],externallyVisible=!!blocker&&blocker.distance>=.12*scale-.00005*scale,compatible=wristNormal.dot(palmNormal)>.45;
    records.push({lane,row,t,point:point.toArray(),palmNormal:palmNormal.toArray(),baselineNormal:wristNormal.toArray(),gap:Math.abs(s.gap),externallyVisible,compatible,blocker:blocker?.object.name??null});
  }}finally{palm.material.side=side;}return records;
}
function palmMetrics(records,wrist){
  const samples=records.filter(s=>s.externallyVisible&&s.compatible),matrix=new THREE.Matrix3().getNormalMatrix(wrist.matrixWorld),n=wrist.geometry.attributes.normal,around=wrist.geometry.parameters.radialSegments;
  check(samples.length>=6,'at least six externally visible compatible palm intersections');
  const rows=samples.map(sample=>{const normal=wristTriangleSample(wrist,sample.lane,sample.row,sample.t,true),palmNormal=new THREE.Vector3().fromArray(sample.palmNormal),old=new THREE.Vector3().fromArray(sample.baselineNormal),angle=a=>Math.acos(Math.max(-1,Math.min(1,a.dot(palmNormal))))*180/Math.PI;return{lane:sample.lane,row:sample.row+sample.t,beforeAngleDegrees:angle(old),candidateAngleDegrees:angle(normal),surfaceGap:sample.gap};});
  const mean=key=>rows.reduce((sum,row)=>sum+row[key],0)/rows.length,beforeMean=mean('beforeAngleDegrees'),candidateMean=mean('candidateAngleDegrees');
  check(candidateMean<beforeMean*.65,'palm seam mean normal mismatch improved by at least 35 percent');
  check(rows.every(row=>row.candidateAngleDegrees<=row.beforeAngleDegrees+1),'every visible compatible palm seam sample retains or improves normal agreement');
  return{samples:rows.length,totalCrossings:records.length,excludedOccluded:records.filter(s=>!s.externallyVisible).length,excludedOpposing:records.filter(s=>s.externallyVisible&&!s.compatible).length,beforeMeanAngleDegrees:beforeMean,candidateMeanAngleDegrees:candidateMean,beforeMaximumAngleDegrees:Math.max(...rows.map(r=>r.beforeAngleDegrees)),candidateMaximumAngleDegrees:Math.max(...rows.map(r=>r.candidateAngleDegrees)),maximumSurfaceGap:Math.max(...rows.map(r=>r.surfaceGap)),rows};
}
function normals(wrist,prior){
  const p=wrist.geometry.attributes.position,n=wrist.geometry.attributes.normal,old=prior.geometry.attributes.normal,idx=wrist.geometry.index.array,matrix=new THREE.Matrix3().getNormalMatrix(wrist.matrix);
  equal(n.count,old.count,'normal attribute count unchanged');let changed=0,rootChanged=0,palmChanged=0,minLength=Infinity,maxLength=0,minimumFaceDot=1,baselineMinimumFaceDot=1,minimumChangedFaceDot=1,minimumUnchangedFaceDot=1;
  const changedRows=new Set(),changedVertices=new Set();
  for(let i=0;i<n.count;i++){const normal=new THREE.Vector3().fromBufferAttribute(n,i);check(normal.toArray().every(Number.isFinite),'finite wrist normal');minLength=Math.min(minLength,normal.length());maxLength=Math.max(maxLength,normal.length());const different=[0,1,2].some(k=>n.array[i*3+k]!==old.array[i*3+k]);if(different){changed++;changedVertices.add(i);changedRows.add(Math.floor(i/wrist.geometry.parameters.radialSegments));if(i<7*wrist.geometry.parameters.radialSegments)rootChanged++;if(i>=8*wrist.geometry.parameters.radialSegments)palmChanged++;}}
  equal(rootChanged,7*wrist.geometry.parameters.radialSegments,'every forearm-side root normal was fitted');check(minLength>.9999&&maxLength<1.0001,'unit wrist normals');
  let baselineNegativeFaces=0,newNegativeFaces=0,unchangedFaces=0;for(let i=0;i<idx.length;i+=3){const ids=Array.from(idx.slice(i,i+3)),[a,b,c]=ids.map(id=>new THREE.Vector3().fromBufferAttribute(p,id).applyMatrix4(wrist.matrix)),face=b.sub(a).cross(c.sub(a)).normalize(),average=new THREE.Vector3(),oldAverage=new THREE.Vector3();for(const id of ids){average.add(new THREE.Vector3().fromBufferAttribute(n,id).applyMatrix3(matrix).normalize());oldAverage.add(new THREE.Vector3().fromBufferAttribute(old,id).applyMatrix3(matrix).normalize());}const dot=face.dot(average.normalize()),oldDot=face.dot(oldAverage.normalize());minimumFaceDot=Math.min(minimumFaceDot,dot);baselineMinimumFaceDot=Math.min(baselineMinimumFaceDot,oldDot);if(oldDot<0){baselineNegativeFaces++;check(dot>=oldDot-1e-7,'pre-existing negative face shading is not worsened');}else{if(dot<=0)newNegativeFaces++;check(dot>0,'no newly reversed wrist face shading');}if(ids.some(id=>changedVertices.has(id)))minimumChangedFaceDot=Math.min(minimumChangedFaceDot,dot);else{unchangedFaces++;equal(dot,oldDot,'unchanged face normal agreement is exact');minimumUnchangedFaceDot=Math.min(minimumUnchangedFaceDot,dot);}}
  return{count:n.count,changed,rootChanged,palmChanged,changedRows:[...changedRows],minLength,maxLength,minimumFaceDot,baselineMinimumFaceDot,minimumChangedFaceDot,minimumUnchangedFaceDot,baselineNegativeFaces,newNegativeFaces,unchangedFaces,unchangedFacesExact:true,SHA256:hash(Buffer.from(n.array.buffer,n.array.byteOffset,n.array.byteLength))};
}
function compare(prior,current){
  equal(fingerprint(current),fingerprint(prior),'complete scene parity except two patient wrist normal arrays');const wrists=[];
  for(const side of ['left','right']){const name='fr-patient-'+side,wrist=current.scene.getObjectByName(name+'-wrist'),priorWrist=prior.scene.getObjectByName(name+'-wrist'),forearm=current.scene.getObjectByName(name+'-forearm'),priorForearm=prior.scene.getObjectByName(name+'-forearm'),priorPalm=prior.scene.getObjectByName(name+'-palm');check(!!wrist&&!!priorWrist&&!!forearm&&!!priorForearm&&!!priorPalm,'patient wrist forearm and palm present');const normal=normals(wrist,priorWrist),samples=sampleCrossings(wrist,forearm),baseline=seamMetrics(samples,priorWrist,priorForearm,false),candidateMetrics=seamMetrics(samples,wrist,forearm,true);check(candidateMetrics.meanAngleDegrees<baseline.meanAngleDegrees*.25,'mean root seam angle reduced by at least 75 percent');check(candidateMetrics.maximumCrossingRootRow<5,'visible root seam remains before palm blend');const crossings=palmCrossings(priorWrist,priorPalm,priorForearm),palm=palmMetrics(crossings,wrist);wrists.push({side,baseline,candidate:candidateMetrics,palm,normals:normal});}return wrists;
}

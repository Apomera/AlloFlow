    async function readingLayout(name,width,scale){
      const reading=await continuation.evaluate(el=>{
        const list=el.querySelector('.circuit-continuation-inquiry-steps'),rect=n=>{const r=n.getBoundingClientRect();return {left:r.left,right:r.right,top:r.top,bottom:r.bottom,width:r.width,height:r.height};};
        const wordsFor=element=>{const walker=document.createTreeWalker(element,NodeFilter.SHOW_TEXT),words=[];let text;while((text=walker.nextNode()))for(const m of text.textContent.matchAll(/\S+/gu)){const range=document.createRange();range.setStart(text,m.index);range.setEnd(text,m.index+m[0].length);const rectangles=[...range.getClientRects()].filter(r=>r.width>.1&&r.height>.1).map(r=>({left:r.left,right:r.right,top:r.top,bottom:r.bottom}));words.push({word:m[0],lineCount:rectangles.reduce((tops,r)=>tops.some(top=>Math.abs(top-r.top)<1)?tops:[...tops,r.top],[]).length,rectangleCount:rectangles.length,rectangles});}return words;};
        return {actualViewportWidth:innerWidth,direction:getComputedStyle(el).direction,listTag:list.tagName,listRole:list.getAttribute('role'),steps:[...list.children].map(li=>{
          const p=li.querySelector('.circuit-continuation-inquiry-prompt'),heading=li.querySelector('.circuit-continuation-inquiry-heading'),number=li.querySelector('.circuit-continuation-inquiry-number'),copy=li.querySelector('.circuit-continuation-inquiry-copy'),style=getComputedStyle(li),card=rect(li);
          return {id:li.dataset.inquiryStep,tag:li.tagName,headingTag:heading.tagName,display:style.display,flexWrap:style.flexWrap,copyDisplay:getComputedStyle(copy).display,card,contentLeft:card.left+parseFloat(style.borderLeftWidth)+parseFloat(style.paddingLeft),contentRight:card.right-parseFloat(style.borderRightWidth)-parseFloat(style.paddingRight),prompt:rect(p),heading:rect(heading),number:rect(number),numberHidden:number.getAttribute('aria-hidden'),words:wordsFor(p),headingWords:wordsFor(heading)};
        })};
      });
      assert.equal(reading.actualViewportWidth,width);assert.equal(reading.listTag,'OL');assert.equal(reading.listRole,'list');assert.deepEqual(reading.steps.map(s=>s.id),['choose','compare']);
      for(const step of reading.steps){assert.equal(step.tag,'LI');assert.equal(step.headingTag,'STRONG');assert.equal(step.numberHidden,'true');for(const word of [...step.words,...step.headingWords])assert.equal(word.lineCount,1,'Word split: '+word.word+' in '+name);assert.equal(step.display,'flex');if(width<=480){assert.equal(step.flexWrap,'wrap');assert.equal(step.copyDisplay,'contents');assert.ok(Math.abs(step.prompt.left-step.contentLeft)<=1);assert.ok(Math.abs(step.prompt.right-step.contentRight)<=1);assert.ok(step.prompt.top>=Math.max(step.number.bottom,step.heading.bottom)-1);assert.ok(step.heading.top>=step.number.top-1);assert.ok(Math.abs(step.number.width-step.number.height)<=1,'Number marker should stay circular at '+scale+'x');}else{assert.notEqual(step.copyDisplay,'contents');}}
      const component=reading.steps.flatMap(s=>s.words).find(w=>w.word==='component');assert.ok(component);assert.equal(component.lineCount,1);results.readingLayouts.push({name,width,scale,...reading});
    }
